Plug & Play

Works With Any LLM

Drop Dolphin into your existing stack in under 5 lines.

Python — OpenAI
from openai import OpenAI
from dolphin_memory import DolphinMemory

client = OpenAI()
memory = DolphinMemory(supabase_url="...", supabase_key="...")

def chat(message, user_id):
    memory.add(message, user_id=user_id)
    context = memory.get_context(message, user_id=user_id)

    return client.chat.completions.create(
        model="gpt-4o",
        messages=[
            {"role": "system", "content": f"User context:\n{context}"},
            {"role": "user", "content": message},
        ]
    ).choices[0].message.content
Python — Google Gemini
import google.generativeai as genai
from dolphin_memory import DolphinMemory

genai.configure(api_key="...")
model = genai.GenerativeModel("gemini-2.5-flash")
memory = DolphinMemory(supabase_url="...", supabase_key="...")

def chat(message, user_id):
    memory.add(message, user_id=user_id)
    context = memory.get_context(message, user_id=user_id)
    return model.generate_content(
        f"User context:\n{context}\n\nUser: {message}"
    )
Python — LangChain
from langchain_openai import ChatOpenAI
from dolphin_memory import DolphinMemory

llm = ChatOpenAI(model="gpt-4o")
memory = DolphinMemory(supabase_url="...", supabase_key="...")

context = memory.get_context("Tell me about myself", user_id="u1")
response = llm.invoke(f"Context:\n{context}\n\nUser: Tell me about myself")
Python — FastAPI
from fastapi import FastAPI
from dolphin_memory import DolphinMemory

app = FastAPI()
memory = DolphinMemory(supabase_url="...", supabase_key="...")
memory.prewarm()  # Load models at startup

@app.post("/chat")
async def chat(message: str, user_id: str):
    memory.add(message, user_id=user_id)
    context = memory.get_context(message, user_id=user_id)
    return {"context": context}  # Pass to your LLM of choice
